W_GAUGE, W_DIA_MM, W_AREA = 0, 2, 6      # column positions used by the program


def currents(i_avg, ripple_pp):
    """Ip and Irms for an inductor carrying i_avg with peak-to-peak ripple."""
    ip = i_avg + ripple_pp / 2
    irms = math.sqrt(i_avg ** 2 + ripple_pp ** 2 / 12)
    return ip, irms


def area_product_cm4(L, ip, irms, bm, j, kw):
    """Ac x Aw = Irms*Ip*L / (Bm*J*Kw) in cm^4.  L in H, Bm in T, J in A/mm^2."""
    bm_cm = bm * 1e-4        # T -> Wb/cm^2
    j_cm = j * 100           # A/mm^2 -> A/cm^2
    return irms * ip * L / (bm_cm * j_cm * kw)


def pick_wire(irms, j):
    """Smallest standard wire whose copper area is equal to or above Irms/J."""
    need = irms / j
    for w in sorted(WIRES, key=lambda w: w[W_AREA]):
        if w[W_AREA] >= need:
            return need, w
    raise ValueError("No wire in the table is large enough; use parallel strands.")


def design(L, i_avg, ripple_pp, bm, j, kw, mlt_mm=None):
    ip, irms = currents(i_avg, ripple_pp)
    ap = area_product_cm4(L, ip, irms, bm, j, kw)
    need, wire = pick_wire(irms, j)
    a_cu = wire[W_AREA]
    notes = []

    for code, le, ae, ve, waac, length, leg, height, al in sorted(CORES, key=lambda c: c[4]):
        if waac < ap:
            continue
        ae_m2 = ae * 1e-6
        n = math.ceil(round(L * ip / (bm * ae_m2), 6))          # N = L*Ip/(Bm*Ac), rounded up
        lg_mm = MU0 * n ** 2 * ae_m2 / L * 1e3                   # lg = mu0*N^2*Ac/L
        aw_win = waac / ae * 1e4                                # window area (mm^2) = WaAc/Ae (cm^4 / mm^2 -> mm^2)
        usable = kw * aw_win
        cu_area = n * a_cu
        if cu_area > usable:
            notes.append(f"E {code}: winding {cu_area:.1f} mm^2 does not fit in usable window "
                         f"{usable:.1f} mm^2 - next core tried.")
            continue
        return dict(ip=ip, irms=irms, ap=ap, code=code, order=f"0_4{code}EC", ae=ae, le=le, ve=ve,
                    waac=waac, dims=(length, leg, height), aw_win=aw_win, n=n, lg_mm=lg_mm,
                    need=need, wire=wire, a_cu=a_cu, j_act=irms / a_cu,
                    bm_act=L * ip / (n * ae_m2), cu_area=cu_area, usable=usable,
                    wire_len_m=(n * mlt_mm / 1000 if mlt_mm else None), notes=notes)
    return dict(ip=ip, irms=irms, ap=ap, code=None, notes=notes)


def ask(prompt, default):
    s = input(f"{prompt} [{default}]: ").strip()
    return float(s) if s else default
